`src/UnityLlmTransport.UnityWebRequest.Core/ChunkQueue.cs`：

```csharp
using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    /// <summary>单写者（Unity 主线程）/单读者（后台）有界字节块队列。</summary>
    public sealed class ChunkQueue
    {
        private readonly object _gate = new object();
        private readonly int _maxBufferedBytes;
        private readonly Queue<byte[]> _chunks = new Queue<byte[]>();
        private int _bufferedBytes;
        private bool _completed;
        private Exception? _completionError;
        private TaskCompletionSource<bool>? _readSignal;

        public ChunkQueue(int maxBufferedBytes)
        {
            if (maxBufferedBytes <= 0)
            {
                throw new ArgumentOutOfRangeException(nameof(maxBufferedBytes));
            }
            _maxBufferedBytes = maxBufferedBytes;
        }

        public int BufferedBytes
        {
            get { lock (_gate) { return _bufferedBytes; } }
        }

        public Exception? CompletionError
        {
            get { lock (_gate) { return _completionError; } }
        }

        public bool TryWrite(byte[] data, int length)
        {
            if (data == null)
            {
                throw new ArgumentNullException(nameof(data));
            }
            if (length < 0 || length > data.Length)
            {
                throw new ArgumentOutOfRangeException(nameof(length));
            }

            TaskCompletionSource<bool>? signal;
            bool ok;
            bool signalValue;
            lock (_gate)
            {
                if (_completed)
                {
                    return false;
                }
                if (length == 0)
                {
                    return true;
                }

                if (_bufferedBytes + length > _maxBufferedBytes)
                {
                    _completed = true;
                    _completionError = new TransportException(
                        TransportErrorKind.Overflow,
                        "Response exceeded ChunkQueue capacity of " + _maxBufferedBytes + " bytes.");
                    _chunks.Clear();
                    _bufferedBytes = 0;
                    ok = false;
                    signalValue = false;
                }
                else
                {
                    var copy = new byte[length];
                    Array.Copy(data, 0, copy, 0, length);
                    _chunks.Enqueue(copy);
                    _bufferedBytes += length;
                    ok = true;
                    signalValue = true;
                }

                signal = TakeSignal();
            }

            if (signal != null)
            {
                signal.TrySetResult(signalValue);
            }
            return ok;
        }

        public bool TryRead(out ReadOnlyMemory<byte> chunk)
        {
            lock (_gate)
            {
                if (_chunks.Count > 0)
                {
                    var next = _chunks.Dequeue();
                    _bufferedBytes -= next.Length;
                    chunk = next;
                    return true;
                }

                chunk = default;
                return false;
            }
        }

        public async ValueTask<bool> WaitToReadAsync(CancellationToken cancellationToken)
        {
            Task<bool> signalTask;
            lock (_gate)
            {
                if (_chunks.Count > 0)
                {
                    return true;
                }
                if (_completed)
                {
                    return false;
                }
                if (_readSignal == null)
                {
                    _readSignal = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
                }
                signalTask = _readSignal.Task;
            }

            return await WaitForSignalAsync(signalTask, cancellationToken).ConfigureAwait(false);
        }

        public void Complete(Exception? error = null)
        {
            TaskCompletionSource<bool>? signal;
            lock (_gate)
            {
                if (_completed)
                {
                    return;
                }
                _completed = true;
                _completionError = error;
                if (error != null)
                {
                    _chunks.Clear();
                    _bufferedBytes = 0;
                }
                signal = TakeSignal();
            }

            if (signal != null)
            {
                signal.TrySetResult(false);
            }
        }

        public void Abort(Exception error)
        {
            if (error == null)
            {
                throw new ArgumentNullException(nameof(error));
            }
            Complete(error);
        }

        private TaskCompletionSource<bool>? TakeSignal()
        {
            var signal = _readSignal;
            _readSignal = null;
            return signal;
        }

        private static async ValueTask<bool> WaitForSignalAsync(Task<bool> signal, CancellationToken cancellationToken)
        {
            if (!cancellationToken.CanBeCanceled)
            {
                return await signal.ConfigureAwait(false);
            }

            var cancelled = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
            using (cancellationToken.Register(() => cancelled.TrySetCanceled(cancellationToken)))
            {
                var completed = await Task.WhenAny(signal, cancelled.Task).ConfigureAwait(false);
                if (completed != signal)
                {
                    cancellationToken.ThrowIfCancellationRequested();
                }
            }

            return await signal.ConfigureAwait(false);
        }
    }
}
```

`src/UnityLlmTransport.UnityWebRequest.Core/ChunkQueueReadStream.cs`：

```csharp
using System;
using System.IO;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    /// <summary>队列→Stream；仅支持异步读（同步读在主线程会卡帧/死锁，明确禁止）。</summary>
    public sealed class ChunkQueueReadStream : Stream
    {
        private readonly ChunkQueue _queue;
        private ReadOnlyMemory<byte> _current;
        private bool _disposed;

        public ChunkQueueReadStream(ChunkQueue queue)
        {
            _queue = queue ?? throw new ArgumentNullException(nameof(queue));
        }

        public override bool CanRead => !_disposed;
        public override bool CanSeek => false;
        public override bool CanWrite => false;
        public override long Length => throw new NotSupportedException();
        public override long Position
        {
            get => throw new NotSupportedException();
            set => throw new NotSupportedException();
        }

        public override int Read(byte[] buffer, int offset, int count)
        {
            throw new NotSupportedException("ChunkQueueReadStream supports async reads only.");
        }

        public override async ValueTask<int> ReadAsync(Memory<byte> buffer, CancellationToken cancellationToken = default)
        {
            if (_disposed)
            {
                throw new ObjectDisposedException(nameof(ChunkQueueReadStream));
            }
            if (buffer.Length == 0)
            {
                return 0;
            }

            while (true)
            {
                if (!_current.IsEmpty)
                {
                    int toCopy = Math.Min(buffer.Length, _current.Length);
                    _current.Slice(0, toCopy).CopyTo(buffer);
                    _current = _current.Slice(toCopy);
                    return toCopy;
                }

                bool available = await _queue.WaitToReadAsync(cancellationToken).ConfigureAwait(false);
                if (!available)
                {
                    var error = _queue.CompletionError;
                    if (error != null)
                    {
                        throw error;
                    }
                    return 0;
                }

                if (!_queue.TryRead(out _current))
                {
                    // 不伪造 EOF：回到 WaitToReadAsync 重新等（单消费者下正常不会走到）。
                }
            }
        }

        public override void Flush() { }
        public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();
        public override void SetLength(long value) => throw new NotSupportedException();
        public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();

        protected override void Dispose(bool disposing)
        {
            if (disposing)
            {
                _disposed = true;
                _current = default;
            }
            base.Dispose(disposing);
        }
    }
}
```

```csharp
using System;

namespace UnityLlmTransport
{
    public static class UwrResponseClassifier
    {
        /// <summary>返回 null = 可交付会话（任意非 2xx 状态）；非 null = 应抛出的 TransportException。</summary>
        public static TransportException? Classify(UwrResponseSnapshot snapshot, bool callerCancelled)
        {
            if (snapshot == null)
            {
                throw new ArgumentNullException(nameof(snapshot));
            }

            if (callerCancelled ||
                snapshot.Completion == UwrCompletion.Aborted ||
                Contains(snapshot.Error, "aborted"))
            {
                return new TransportException(
                    TransportErrorKind.Cancelled,
                    "Request cancelled.",
                    null,
                    snapshot.StatusCode,
                    snapshot.Headers,
                    snapshot.Error,
                    false);
            }

            if (snapshot.StatusCode > 0)
            {
                return null;
            }

            var error = snapshot.Error ?? string.Empty;
            if (snapshot.Completion == UwrCompletion.ConnectionError ||
                snapshot.Completion == UwrCompletion.ProtocolError)
            {
                if (Contains(error, "SSL") || Contains(error, "certificate"))
                {
                    return new TransportException(
                        TransportErrorKind.Tls,
                        "TLS handshake failed: " + error,
                        null,
                        null,
                        snapshot.Headers,
                        error,
                        false);
                }
                if (Contains(error, "proxy") || Contains(error, "Tunnel") || Contains(error, "407"))
                {
                    return new TransportException(
                        TransportErrorKind.Proxy,
                        "Proxy failure: " + error,
                        null,
                        null,
                        snapshot.Headers,
                        error,
                        false);
                }
                return new TransportException(
                    TransportErrorKind.Connect,
                    "Connection failed: " + error,
                    null,
                    null,
                    snapshot.Headers,
                    error,
                    true);
            }

            if (snapshot.Completion == UwrCompletion.DataProcessingError)
            {
                return new TransportException(
                    TransportErrorKind.Protocol,
                    "Data processing failed: " + error,
                    null,
                    null,
                    snapshot.Headers,
                    error,
                    false);
            }

            return new TransportException(
                TransportErrorKind.Protocol,
                "Success without HTTP status code.",
                null,
                null,
                snapshot.Headers,
                error,
                false);
        }

        private static bool Contains(string? value, string fragment)
        {
            return value != null && value.IndexOf(fragment, StringComparison.OrdinalIgnoreCase) >= 0;
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;

namespace UnityLlmTransport
{
    public static class UwrRequestPlanner
    {
        /// <summary>纯决策：能力不支持抛 Unsupported，不静默降级。</summary>
        public static UwrRequestPlan Create(TransportRequest request, TransportOptions options)
        {
            if (request == null)
            {
                throw new ArgumentNullException(nameof(request));
            }
            if (options == null)
            {
                throw new ArgumentNullException(nameof(options));
            }

            if (request.BodyFactory != null)
            {
                throw new TransportException(
                    TransportErrorKind.Unsupported,
                    "UWR does not support streaming request bodies (BodyFactory).");
            }

            if (options.Proxy != null && options.Proxy.Mode != ProxyMode.System)
            {
                throw new TransportException(
                    TransportErrorKind.Unsupported,
                    "UWR supports only the system proxy (ProxyMode." + options.Proxy.Mode + " is unsupported).");
            }

            bool acceptAll = false;
            if (options.Tls != null)
            {
                if (options.Tls.ClientCertificate != null ||
                    options.Tls.CustomRoots != null ||
                    options.Tls.Protocols != null)
                {
                    throw new TransportException(
                        TransportErrorKind.Unsupported,
                        "UWR supports only default TLS validation or accept-all; custom certificate/protocols are unsupported.");
                }
                acceptAll = !options.Tls.ValidateCertificate;
            }

            int redirectLimit = options.Redirects == RedirectPolicy.Error ? 0 : 32;

            var headers = new TransportHeaders();
            var suppressed = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (var pair in request.Headers)
            {
                if (pair.Value == null)
                {
                    suppressed.Add(pair.Key);
                }
                else
                {
                    headers[pair.Key] = pair.Value;
                }
            }

            if (options.Decompress &&
                !headers.ContainsKey("Accept-Encoding") &&
                !suppressed.Contains("Accept-Encoding"))
            {
                headers["Accept-Encoding"] = "identity";
            }

            return new UwrRequestPlan(headers, redirectLimit, acceptAll);
        }
    }
}
```

```csharp
using System;
```

```csharp
namespace UnityLlmTransport
{
    public sealed class UnityLifecycleToken
    {
        private CancellationTokenSource _cts = new CancellationTokenSource();
        private int _disposed;

        public CancellationToken Token
        {
            get { return Volatile.Read(ref _cts).Token; }
        }

        public void Invalidate()
        {
            if (Volatile.Read(ref _disposed) != 0)
            {
                return;
            }
            var next = new CancellationTokenSource();
            var previous = Interlocked.Exchange(ref _cts, next);
            previous.Cancel();
            previous.Dispose();
        }

        public void Dispose()
        {
            if (Interlocked.Exchange(ref _disposed, 1) != 0)
            {
                return;
            }
            Volatile.Read(ref _cts).Cancel();
        }
    }
```

```csharp
using System;
using System.Collections;
using System.Threading;
using System.Threading.Tasks;
using UnityEngine;
using UnityEngine.Networking;

namespace UnityLlmTransport
{
    public sealed class UnityCoroutineHost : MonoBehaviour, IUnityWebRequestRunner
    {
        private readonly UnityMainThreadDispatcher _dispatcher = new UnityMainThreadDispatcher();
        private readonly UnityLifecycleToken _lifecycle = new UnityLifecycleToken();

        public CancellationToken ShutdownToken
        {
            get { return _lifecycle.Token; }
        }

        private void Awake()
        {
            DontDestroyOnLoad(gameObject);
        }

        private void Update()
        {
            _dispatcher.Drain();
        }

        private void OnDestroy()
        {
            _lifecycle.Dispose();
        }

        private void OnApplicationQuit()
        {
            _lifecycle.Dispose();
        }

        public void InvalidateLifecycle()
        {
            _lifecycle.Invalidate();
        }

        public void Post(Action action)
        {
            _dispatcher.Post(action);
        }

        public Task<T> InvokeAsync<T>(Func<T> func, CancellationToken ct)
        {
            if (func == null)
            {
                throw new ArgumentNullException(nameof(func));
            }

            var completion = new TaskCompletionSource<T>(TaskCreationOptions.RunContinuationsAsynchronously);
            var registration = ct.Register(() => completion.TrySetCanceled(ct));
            completion.Task.ContinueWith(_ => registration.Dispose(), TaskScheduler.Default);

            _dispatcher.Post(() =>
            {
                if (ct.IsCancellationRequested)
                {
                    completion.TrySetCanceled(ct);
                    return;
                }
                try
                {
                    completion.TrySetResult(func());
                }
                catch (Exception ex)
                {
                    completion.TrySetException(ex);
                }
            });

            return completion.Task;
        }

        public Task<UwrResponseSnapshot> StartAsync(UnityWebRequest request, CancellationToken ct)
        {
            if (request == null)
            {
                throw new ArgumentNullException(nameof(request));
            }

            var completion = new TaskCompletionSource<UwrResponseSnapshot>(TaskCreationOptions.RunContinuationsAsynchronously);
            _dispatcher.Post(() => StartCoroutine(RunRequest(request, ct, completion)));
            return completion.Task;
        }

        private IEnumerator RunRequest(
            UnityWebRequest request,
            CancellationToken ct,
            TaskCompletionSource<UwrResponseSnapshot> completion)
        {
            if (ct.IsCancellationRequested)
            {
                completion.TrySetResult(new UwrResponseSnapshot(
                    UwrCompletion.Aborted,
                    0,
                    new TransportHeaders(),
                    "Cancelled before start."));
                yield break;
            }

            int cancelled = 0;
            CancellationTokenRegistration registration = default;
            if (ct.CanBeCanceled)
            {
                registration = ct.Register(() =>
                {
                    Interlocked.Exchange(ref cancelled, 1);
                    _dispatcher.Post(() => request.Abort());
                });
            }

            UnityWebRequestAsyncOperation operation = request.SendWebRequest();
            IStreamingDownloadHandler handler = request.downloadHandler as IStreamingDownloadHandler;

            while (!operation.isDone)
            {
                if (handler != null && handler.FirstDataReceived && request.responseCode != 0)
                {
                    break;
                }
                yield return null;
            }

            var snapshot = BuildSnapshot(request, Volatile.Read(ref cancelled) != 0);
            registration.Dispose();
            completion.TrySetResult(snapshot);
        }

        private static UwrResponseSnapshot BuildSnapshot(UnityWebRequest request, bool cancelled)
        {
            var completion = cancelled ? UwrCompletion.Aborted : MapResult(request.result);
            var headers = new TransportHeaders();
            try
            {
                var values = request.GetResponseHeaders();
                if (values != null)
                {
                    foreach (var pair in values)
                    {
                        headers[pair.Key] = pair.Value;
                    }
                }
            }
            catch (Exception)
            {
                // 响应头尚未可用时容忍为空；U2 门禁负责验证首字节时 headers 可用。
            }

            return new UwrResponseSnapshot(completion, (int)request.responseCode, headers, request.error);
        }

        private static UwrCompletion MapResult(UnityWebRequest.Result result)
        {
            switch (result)
            {
                case UnityWebRequest.Result.InProgress:
                    return UwrCompletion.InProgress;
                case UnityWebRequest.Result.Success:
                    return UwrCompletion.Success;
                case UnityWebRequest.Result.ConnectionError:
                    return UwrCompletion.ConnectionError;
                case UnityWebRequest.Result.ProtocolError:
                    return UwrCompletion.ProtocolError;
                case UnityWebRequest.Result.DataProcessingError:
                    return UwrCompletion.DataProcessingError;
                default:
                    return UwrCompletion.ProtocolError;
            }
        }
    }
}
```

```csharp
using System;
using UnityEngine.Networking;

namespace UnityLlmTransport
{
    public sealed class StreamingDownloadHandler : DownloadHandlerScript, IStreamingDownloadHandler
    {
        private readonly ChunkQueue _queue;
        private volatile bool _aborted;

        public StreamingDownloadHandler(ChunkQueue queue, int downloadBufferSize)
            : base(new byte[downloadBufferSize])
        {
            if (queue == null)
            {
                throw new ArgumentNullException(nameof(queue));
            }
            if (downloadBufferSize <= 0)
            {
                throw new ArgumentOutOfRangeException(nameof(downloadBufferSize));
            }
            _queue = queue;
        }

        public bool FirstDataReceived { get; private set; }

        protected override bool ReceiveData(byte[] data, int dataLength)
        {
            if (_aborted)
            {
                return false;
            }
            if (dataLength <= 0)
            {
                return true;
            }
            if (!_queue.TryWrite(data, dataLength))
            {
                _aborted = true;
                return false;
            }
            FirstDataReceived = true;
            return true;
        }

        protected override void CompleteContent()
        {
            _queue.Complete();
        }

        public void MarkAborted(TransportErrorKind kind)
        {
            if (_aborted)
            {
                return;
            }
            _aborted = true;
            _queue.Abort(new TransportException(kind, "Response aborted (" + kind + ")."));
        }
    }
}
```

```csharp
using System;
using System.IO;
using System.IO.Compression;
using System.Threading;
using System.Threading.Tasks;
using UnityEngine.Networking;

namespace UnityLlmTransport
{
    public sealed class UnityWebRequestSession : ITransportSession
    {
        private readonly UnityWebRequest _request;
        private readonly StreamingDownloadHandler _handler;
        private readonly ChunkQueue _queue;
        private readonly IUnityWebRequestRunner _runner;
        private readonly CancellationTokenRegistration _cancellationRegistration;
        private int _disposed;

        internal UnityWebRequestSession(
            UnityWebRequest request,
            StreamingDownloadHandler handler,
            ChunkQueue queue,
            int statusCode,
            TransportHeaders headers,
            bool decompress,
            IUnityWebRequestRunner runner,
            CancellationToken lifetimeToken)
        {
            _request = request;
            _handler = handler;
            _queue = queue;
            _runner = runner;
            StatusCode = statusCode;
            Headers = headers;

            Stream body = new ChunkQueueReadStream(queue);
            if (decompress && IsGzip(headers))
            {
                body = new GZipStream(body, CompressionMode.Decompress, leaveOpen: false);
            }
            Body = body;

            _cancellationRegistration = lifetimeToken.Register(AbortFromCancellation);
        }

        public int StatusCode { get; }
        public TransportHeaders Headers { get; }
        public Stream Body { get; }

        public void Abort()
        {
            AbortWith(TransportErrorKind.Aborted);
        }

        public ValueTask DisposeAsync()
        {
            if (Interlocked.Exchange(ref _disposed, 1) == 0)
            {
                AbortWith(TransportErrorKind.Aborted);
                _cancellationRegistration.Dispose();
                _runner.Post(() => _request.Dispose());
            }
            return default;
        }

        private void AbortFromCancellation()
        {
            AbortWith(TransportErrorKind.Cancelled);
        }

        private void AbortWith(TransportErrorKind kind)
        {
            _handler.MarkAborted(kind);
            _queue.Abort(new TransportException(kind, "Session aborted (" + kind + ")."));
            _runner.Post(() => _request.Abort());
        }

        private static bool IsGzip(TransportHeaders headers)
        {
            return headers.TryGetValue("Content-Encoding", out var value) &&
                   value != null &&
                   value.IndexOf("gzip", StringComparison.OrdinalIgnoreCase) >= 0;
        }
    }
}
```

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;
using UnityEngine.Networking;

namespace UnityLlmTransport
{
    public sealed class UnityWebRequestTransport : IStreamingHttpTransport
    {
        private readonly IUnityWebRequestRunner _runner;
        private readonly UnityWebRequestTransportOptions _options;

        public UnityWebRequestTransport(
            IUnityWebRequestRunner runner,
            UnityWebRequestTransportOptions options = null)
        {
            _runner = runner ?? throw new ArgumentNullException(nameof(runner));
            _options = options ?? new UnityWebRequestTransportOptions();
        }

        public TransportCapabilities Capabilities =>
            TransportCapabilities.StreamingBody |
            TransportCapabilities.SystemProxy |
            TransportCapabilities.TlsCertificateHandler |
            TransportCapabilities.Compression;

        public async Task<ITransportSession> OpenAsync(
            TransportRequest request,
            TransportOptions options,
            CancellationToken cancellationToken)
        {
            var plan = UwrRequestPlanner.Create(request, options);
            var queue = new ChunkQueue(_options.MaxBufferedBytes);
            UnityWebRequest requestToDispose = null;

            try
            {
                var built = await _runner.InvokeAsync(
                    () => BuildRequest(request, options, plan, queue),
                    cancellationToken).ConfigureAwait(false);
                var uwr = built.Request;
                requestToDispose = uwr;

                UwrResponseSnapshot snapshot = await StartWithFirstByteTimeoutAsync(
                    uwr, options, cancellationToken, queue).ConfigureAwait(false);

                TransportException failure;
                if (plan.RedirectLimit == 0 && snapshot.StatusCode >= 300 && snapshot.StatusCode < 400)
                {
                    failure = new TransportException(
                        TransportErrorKind.Protocol,
                        "Redirect not allowed by RedirectPolicy.Error.",
                        null,
                        snapshot.StatusCode,
                        snapshot.Headers,
                        snapshot.Error,
                        false);
                }
                else
                {
                    failure = UwrResponseClassifier.Classify(snapshot, cancellationToken.IsCancellationRequested);
                }

                if (failure != null)
                {
                    queue.Abort(failure);
                    var disposeThis = uwr;
                    requestToDispose = null;
                    _runner.Post(() =>
                    {
                        disposeThis.Abort();
                        disposeThis.Dispose();
                    });
                    throw failure;
                }

                return new UnityWebRequestSession(
                    uwr,
                    built.Handler,
                    queue,
                    snapshot.StatusCode,
                    snapshot.Headers,
                    options.Decompress,
                    _runner,
                    cancellationToken);
            }
            catch
            {
                if (requestToDispose != null)
                {
                    var disposeThis = requestToDispose;
                    _runner.Post(() =>
                    {
                        disposeThis.Abort();
                        disposeThis.Dispose();
                    });
                }
                throw;
            }
        }

        private BuiltRequest BuildRequest(
            TransportRequest request,
            TransportOptions options,
            UwrRequestPlan plan,
            ChunkQueue queue)
        {
            var uwr = new UnityWebRequest(request.Url, request.Method);
            if (request.Body.Length > 0)
            {
                uwr.uploadHandler = new UploadHandlerRaw(request.Body.ToArray());
            }

            var handler = new StreamingDownloadHandler(queue, _options.DownloadBufferSize);
            uwr.downloadHandler = handler;
            uwr.timeout = options.ConnectTimeout.HasValue && options.ConnectTimeout.Value > TimeSpan.Zero
                ? (int)Math.Ceiling(options.ConnectTimeout.Value.TotalSeconds)
                : 0;
            uwr.redirectLimit = plan.RedirectLimit;
            if (plan.AcceptAllCertificates)
            {
                uwr.certificateHandler = new AcceptAllCertificateHandler();
            }
            foreach (var pair in plan.Headers)
            {
                uwr.SetRequestHeader(pair.Key, pair.Value);
            }

            return new BuiltRequest(uwr, handler);
        }

        private async Task<UwrResponseSnapshot> StartWithFirstByteTimeoutAsync(
            UnityWebRequest uwr,
            TransportOptions options,
            CancellationToken cancellationToken,
            ChunkQueue queue)
        {
            CancellationTokenSource timeoutCts = null;
            CancellationTokenSource linkedCts = null;
            try
            {
                CancellationToken token = cancellationToken;
                var firstByteTimeout = options.FirstByteTimeout;
                if (firstByteTimeout.HasValue && firstByteTimeout.Value > TimeSpan.Zero)
                {
                    timeoutCts = new CancellationTokenSource(firstByteTimeout.Value);
                    linkedCts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken, timeoutCts.Token);
                    token = linkedCts.Token;
                }

                var snapshot = await _runner.StartAsync(uwr, token).ConfigureAwait(false);
                if (timeoutCts != null && timeoutCts.IsCancellationRequested && snapshot.Completion == UwrCompletion.Aborted)
                {
                    var timeoutError = new TransportException(TransportErrorKind.FirstByteTimeout, "First byte timeout.");
                    queue.Abort(timeoutError);
                    throw timeoutError;
                }
                return snapshot;
            }
            finally
            {
                linkedCts?.Dispose();
                timeoutCts?.Dispose();
            }
        }

        private sealed record BuiltRequest(UnityWebRequest Request, StreamingDownloadHandler Handler);
    }
}
```

```csharp
using System;
using System.Collections;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Text;
using System.Threading;
using System.Threading.Tasks;
using UnityLlmTransport;
using UnityLlmTransport.Sse;
using UnityEngine;

public sealed class Phase2Runner : MonoBehaviour
{
    private readonly StringBuilder _results = new StringBuilder();
    private UnityCoroutineHost _host;
    private UnityWebRequestTransport _transport;
    private int _mainThreadId;

    private IEnumerator Start()
    {
        _mainThreadId = Thread.CurrentThread.ManagedThreadId;
        _results.AppendLine("unity_version=" + Application.unityVersion);
        _results.AppendLine("main_thread=" + _mainThreadId);

        var hostObject = new GameObject("Phase2Host");
        _host = hostObject.AddComponent<UnityCoroutineHost>();
        yield return null;

        _transport = new UnityWebRequestTransport(_host, new UnityWebRequestTransportOptions());
        yield return Run("U1_stream", ScenarioU1);
        yield return Run("U2_snapshot", ScenarioU2);

        var path = Path.Combine(Application.persistentDataPath, "phase2-results.txt");
        File.WriteAllText(path, _results.ToString());
        UnityEngine.Debug.Log("PHASE2_RESULTS_PATH=" + path);
        foreach (var line in _results.ToString().Split('\n'))
        {
            UnityEngine.Debug.Log("PHASE2 " + line);
        }
        yield return new WaitForSeconds(1f);
        Application.Quit();
    }

    private IEnumerator Run(string name, Func<Task<string>> scenario)
    {
        Task<string> task = scenario();
        while (!task.IsCompleted)
        {
            yield return null;
        }

        if (task.IsFaulted)
        {
            Exception error = task.Exception?.InnerException ?? task.Exception;
            _results.AppendLine(name + "|FAIL|" + Describe(error));
        }
        else if (task.IsCanceled)
        {
            _results.AppendLine(name + "|FAIL|cancelled");
        }
        else
        {
            _results.AppendLine(name + "|OK|" + task.Result);
        }
    }

    private static string Describe(Exception error)
    {
        if (error == null)
        {
            return "none";
        }
        if (error is TransportException transport)
        {
            return "TransportException kind=" + transport.Kind +
                   " code=" + (transport.StatusCode.HasValue ? transport.StatusCode.Value.ToString() : "-") +
                   " msg=" + Trunc(transport.Message, 80);
        }
        return error.GetType().Name + ": " + Trunc(error.Message, 80);
    }

    private async Task<string> ScenarioU1()
    {
        var sw = Stopwatch.StartNew();
        var session = await _transport.OpenAsync(Request("/sse"), new TransportOptions(), CancellationToken.None);
        long openMs = sw.ElapsedMilliseconds;
        var parse = await ParseAsync(session.Body, sw);
        await session.DisposeAsync();
        return "status=" + session.StatusCode +
               "|open_ms=" + openMs +
               "|items=" + parse.Items.Count +
               "|first_ms=" + parse.FirstItemMs +
               "|total_ms=" + sw.ElapsedMilliseconds +
               "|data=" + string.Join("", parse.Items);
    }

    private async Task<string> ScenarioU2()
    {
        var session = await _transport.OpenAsync(Request("/sse"), new TransportOptions(), CancellationToken.None);
        bool hasContentType = session.Headers.TryGetValue("content-type", out var contentType) &&
                              contentType != null &&
                              contentType.Contains("text/event-stream");
        var buffer = new byte[64];
        await session.Body.ReadAsync(buffer.AsMemory(), CancellationToken.None);
        await session.DisposeAsync();
        return "status=" + session.StatusCode + "|content_type=" + hasContentType;
    }

    private static async Task<ParseResult> ParseAsync(Stream stream, Stopwatch sw)
    {
        var items = new List<string>();
        long firstMs = -1;
        await foreach (var item in SseParser.Create(stream).EnumerateAsync())
        {
            if (firstMs < 0)
            {
                firstMs = sw.ElapsedMilliseconds;
            }
            items.Add(item.Data);
        }
        return new ParseResult(items, firstMs);
    }

    private static TransportRequest Request(string path)
    {
        return new TransportRequest(
            "POST",
            "http://127.0.0.1:5199" + path,
            new TransportHeaders
            {
                ["accept"] = "text/event-stream",
                ["content-type"] = "application/json"
            },
            new ReadOnlyMemory<byte>(Encoding.UTF8.GetBytes("{}")));
    }

    private static string Trunc(string value, int max)
    {
        if (string.IsNullOrEmpty(value))
        {
            return string.Empty;
        }
        var singleLine = value.Replace('\n', ' ').Replace('\r', ' ');
        return singleLine.Length <= max ? singleLine : singleLine.Substring(0, max) + "...";
    }

    private sealed class ParseResult
    {
        public ParseResult(List<string> items, long firstItemMs)
        {
            Items = items;
            FirstItemMs = firstItemMs;
        }

        public List<string> Items { get; }
        public long FirstItemMs { get; }
    }
}
```

```csharp
Console.WriteLine("SSE server: /sse | /sse-error | /sse-gzip | /sse-slow | /sse-idle | /sse-big | /redirect");

while (true)
{
    var ctx = await listener.GetContextAsync();
    var path = ctx.Request.Url!.AbsolutePath;
    if (path is not ("/sse" or "/sse-error" or "/sse-gzip" or "/sse-slow" or "/sse-idle" or "/sse-big" or "/redirect"))
    {
        ctx.Response.StatusCode = 404;
        ctx.Response.Close();
        continue;
    }

    try
    {
        if (path == "/redirect")
        {
            ctx.Response.StatusCode = 302;
            ctx.Response.Headers["Location"] = "/sse";
            ctx.Response.Close();
            continue;
        }

        ctx.Response.ContentType = "text/event-stream";
        ctx.Response.SendChunked = true;
        var output = ctx.Response.OutputStream;

        if (path == "/sse-error")
        {
            ctx.Response.StatusCode = 500;
            await output.WriteAsync(Encoding.UTF8.GetBytes("{\"error\":\"boom\"}"));
            ctx.Response.Close();
            continue;
        }

        if (path == "/sse-big")
        {
            var payload = Encoding.UTF8.GetBytes("data: " + new string('x', 512) + "\n\n");
            for (int i = 0; i < 4096; i++)
            {
                await output.WriteAsync(payload);
                await output.FlushAsync();
            }
            await output.WriteAsync(Encoding.UTF8.GetBytes("data: [DONE]\n\n"));
            ctx.Response.Close();
            continue;
        }

        if (path == "/sse-slow")
        {
            await Task.Delay(600);
        }

        if (path == "/sse-gzip")
        {
            ctx.Response.Headers["Content-Encoding"] = "gzip";
            using (var gzip = new GZipStream(output, CompressionLevel.Fastest, leaveOpen: true))
            {
                await WritePartsAsync(gzip, 100);
                await gzip.WriteAsync(Encoding.UTF8.GetBytes("data: [DONE]\n\n"));
                await gzip.FlushAsync();
            }
            await output.FlushAsync();
            ctx.Response.Close();
            continue;
        }

        await WritePartsAsync(output, path == "/sse-idle" ? 400 : 100);
        await output.WriteAsync(Encoding.UTF8.GetBytes("data: [DONE]\n\n"));
        await output.FlushAsync();
        ctx.Response.Close();
    }
    catch (Exception ex)
    {
        Console.WriteLine("request error: " + ex.Message);
        try { ctx.Response.Abort(); } catch { }
    }
}

static async Task WritePartsAsync(Stream output, int delayMs)
{
    foreach (var part in new[] { "你", "好", "，", "世", "界" })
    {
        var payload = $"data: {{\"choices\":[{{\"delta\":{{\"content\":\"{part}\"}}}}]}}\n\n";
        await output.WriteAsync(Encoding.UTF8.GetBytes(payload));
        await output.FlushAsync();
        await Task.Delay(delayMs);
    }
}
```
