**Goal:** 落地纯 .NET 的通讯契约与核心件（Abstractions + SSE + 装饰器 + Mock），让 Phase 2+ 的 UWR/HttpClient/WebSocket 适配器与 PiUnityAgent.Ai 只依赖这些契约。

**Tech Stack:** C# 9、netstandard2.1（两个库）、net10.0 xUnit（测试）、vendor System.Net.ServerSentEvents (MIT)。

```csharp
namespace UnityLlmTransport.Tests
```

```csharp
            int read;

            {
```

```csharp
using System;
using System.Collections.Generic;
using System.IO;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    internal sealed class ScriptedResponseStream : Stream
    {
        private readonly List<ReadOnlyMemory<byte>> _chunks;
        private readonly TimeSpan _chunkDelay;
        private int _chunkIndex;
        private int _offset;
        private bool _aborted;

        public ScriptedResponseStream(
            IReadOnlyList<ReadOnlyMemory<byte>>? chunks,
            TimeSpan chunkDelay,
            CancellationToken cancellationToken)
        {
            _chunks = new List<ReadOnlyMemory<byte>>();
            if (chunks != null)
            {
                for (int i = 0; i < chunks.Count; i++)
                {
                    if (chunks[i].Length > 0)
                    {
                        _chunks.Add(chunks[i]);
                    }
                }
            }
            _chunkDelay = chunkDelay;
        }

        public void Abort()
        {
            _aborted = true;
        }

        public override bool CanRead => true;
        public override bool CanSeek => false;
        public override bool CanWrite => false;
        public override long Length => throw new NotSupportedException();
        public override long Position
        {
            get => throw new NotSupportedException();
            set => throw new NotSupportedException();
        }

        public override void Flush() { }
        public override int Read(byte[] buffer, int offset, int count) => throw new NotSupportedException("Use ReadAsync.");
        public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();
        public override void SetLength(long value) => throw new NotSupportedException();
        public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();

        public override async ValueTask<int> ReadAsync(Memory<byte> buffer, CancellationToken cancellationToken = default)
        {
            if (_aborted || _chunkIndex >= _chunks.Count)
            {
                return 0;
            }

            if (buffer.Length == 0)
            {
                return 0;
            }

            if (_chunkDelay > TimeSpan.Zero)
            {
                await Task.Delay(_chunkDelay, cancellationToken).ConfigureAwait(false);
            }

            if (_aborted)
            {
                return 0;
            }

            var chunk = _chunks[_chunkIndex];
            int remaining = chunk.Length - _offset;
            int toCopy = remaining < buffer.Length ? remaining : buffer.Length;
            chunk.Slice(_offset, toCopy).CopyTo(buffer);
            _offset += toCopy;
            if (_offset == chunk.Length)
            {
                _chunkIndex++;
                _offset = 0;
            }
            return toCopy;
        }
    }
}
```

```csharp
using System;
using System.IO;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    /// <summary>守卫连接后的首字节、流空闲与响应体大小；超时/超限即 Abort 内层会话。</summary>
    public sealed class TimeoutTransport : IStreamingHttpTransport
    {
        private readonly IStreamingHttpTransport _inner;

        public TimeoutTransport(IStreamingHttpTransport inner)
        {
            _inner = inner ?? throw new ArgumentNullException(nameof(inner));
        }

        public TransportCapabilities Capabilities => _inner.Capabilities;

        public async Task<ITransportSession> OpenAsync(
            TransportRequest request,
            TransportOptions options,
            CancellationToken cancellationToken)
        {
            var session = await _inner.OpenAsync(request, options, cancellationToken).ConfigureAwait(false);
            return new TimedSession(session, options);
        }

        private sealed class TimedSession : ITransportSession
        {
            private readonly ITransportSession _inner;
            private readonly bool _hasFirstByteBudget;
            private readonly DateTimeOffset _firstByteDeadline;
            private readonly TimeSpan _idleTimeout;
            private readonly int _maxResponseBytes;
            private readonly TimedBodyStream _body;
            private long _receivedBytes;

            public TimedSession(ITransportSession inner, TransportOptions options)
            {
                _inner = inner;
                _hasFirstByteBudget = options.FirstByteTimeout.HasValue;
                _firstByteDeadline = _hasFirstByteBudget
                    ? DateTimeOffset.UtcNow + options.FirstByteTimeout!.Value
                    : DateTimeOffset.MaxValue;
                _idleTimeout = options.IdleTimeout ?? Timeout.InfiniteTimeSpan;
                _maxResponseBytes = options.MaxResponseBytes ?? -1;
                _body = new TimedBodyStream(this, inner.Body);
            }

            public int StatusCode => _inner.StatusCode;
            public TransportHeaders Headers => _inner.Headers;
            public Stream Body => _body;
            public bool FirstByteReceived { get; private set; }

            public void Abort() => _inner.Abort();

            public ValueTask DisposeAsync() => _inner.DisposeAsync();

            public TimeSpan GetReadBudget()
            {
                if (!FirstByteReceived)
                {
                    if (!_hasFirstByteBudget)
                    {
                        return Timeout.InfiniteTimeSpan;
                    }
                    var remaining = _firstByteDeadline - DateTimeOffset.UtcNow;
                    return remaining > TimeSpan.Zero ? remaining : TimeSpan.Zero;
                }
                return _idleTimeout;
            }

            public void MarkFirstByte()
            {
                FirstByteReceived = true;
            }

            public void AddReceivedBytes(int count)
            {
                _receivedBytes += count;
                if (_maxResponseBytes >= 0 && _receivedBytes > _maxResponseBytes)
                {
                    _inner.Abort();
                    throw new TransportException(
                        TransportErrorKind.Overflow,
                        "Response exceeded MaxResponseBytes (" + _maxResponseBytes + ").");
                }
            }

            public void AbortForTimeout(TransportErrorKind kind)
            {
                _inner.Abort();
            }
        }

        private sealed class TimedBodyStream : Stream
        {
            private readonly TimedSession _session;
            private readonly Stream _inner;

            public TimedBodyStream(TimedSession session, Stream inner)
            {
                _session = session;
                _inner = inner;
            }

            public override bool CanRead => true;
            public override bool CanSeek => false;
            public override bool CanWrite => false;
            public override long Length => throw new NotSupportedException();
            public override long Position
            {
                get => throw new NotSupportedException();
                set => throw new NotSupportedException();
            }

            public override void Flush() { }
            public override int Read(byte[] buffer, int offset, int count) => throw new NotSupportedException("Use ReadAsync.");
            public override long Seek(long offset, SeekOrigin origin) => throw new NotSupportedException();
            public override void SetLength(long value) => throw new NotSupportedException();
            public override void Write(byte[] buffer, int offset, int count) => throw new NotSupportedException();

            public override async ValueTask<int> ReadAsync(Memory<byte> buffer, CancellationToken cancellationToken = default)
            {
                var budget = _session.GetReadBudget();
                if (budget == TimeSpan.Zero)
                {
                    var kindBeforeRead = _session.FirstByteReceived
                        ? TransportErrorKind.IdleTimeout
                        : TransportErrorKind.FirstByteTimeout;
                    _session.AbortForTimeout(kindBeforeRead);
                    throw new TransportException(kindBeforeRead, "Read timed out.");
                }

                CancellationTokenSource? timeoutCts = null;
                CancellationTokenSource? linkedCts = null;
                try
                {
                    if (budget != Timeout.InfiniteTimeSpan)
                    {
                        timeoutCts = new CancellationTokenSource(budget);
                        linkedCts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken, timeoutCts.Token);
                    }

                    var token = linkedCts != null ? linkedCts.Token : cancellationToken;
                    int read;
                    try
                    {
                        read = await _inner.ReadAsync(buffer, token).ConfigureAwait(false);
                    }
                    catch (OperationCanceledException ex) when (
                        !cancellationToken.IsCancellationRequested &&
                        timeoutCts != null &&
                        timeoutCts.IsCancellationRequested)
                    {
                        var kind = _session.FirstByteReceived
                            ? TransportErrorKind.IdleTimeout
                            : TransportErrorKind.FirstByteTimeout;
                        _session.AbortForTimeout(kind);
                        throw new TransportException(kind, "Read timed out.", ex);
                    }

                    if (read == 0)
                    {
                        return 0;
                    }

                    if (!_session.FirstByteReceived)
                    {
                        _session.MarkFirstByte();
                    }

                    _session.AddReceivedBytes(read);
                    return read;
                }
                finally
                {
                    if (linkedCts != null)
                    {
                        linkedCts.Dispose();
                    }
                    if (timeoutCts != null)
                    {
                        timeoutCts.Dispose();
                    }
                }
            }
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    public interface IRequestMiddleware
    {
        ValueTask<TransportRequest> InvokeAsync(TransportRequest request, CancellationToken ct);
    }

    public sealed class MiddlewareTransport : IStreamingHttpTransport
    {
        private readonly IStreamingHttpTransport _inner;
        private readonly IReadOnlyList<IRequestMiddleware> _middlewares;

        public MiddlewareTransport(IStreamingHttpTransport inner, IEnumerable<IRequestMiddleware> middlewares)
        {

        public async Task<ITransportSession> OpenAsync(
            TransportRequest request,
            TransportOptions options,
            CancellationToken cancellationToken)
        {
            var current = request;
            for (int i = 0; i < _middlewares.Count; i++)
            {
                current = await _middlewares[i].InvokeAsync(current, cancellationToken).ConfigureAwait(false);
            }
            return await _inner.OpenAsync(current, options, cancellationToken).ConfigureAwait(false);
        }
    }
```

```csharp
using System;
using System.Globalization;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    public sealed record RetryOptions
    {
        public int MaxRetries { get; init; } = 2;
        public TimeSpan MaxRetryDelay { get; init; } = TimeSpan.FromSeconds(60);
        public TimeSpan BaseDelay { get; init; } = TimeSpan.FromMilliseconds(500);
    }

    public interface IRetryPolicy
    {
        bool ShouldRetry(TransportException error, int attempt);
        TimeSpan GetDelay(TransportException error, int attempt);
    }

    public interface IAuthRefreshHandler
    {
        Task<bool> RefreshAsync(CancellationToken ct);
    }

    public sealed class RetryTransport : IStreamingHttpTransport
    {
        private readonly IStreamingHttpTransport _inner;
        private readonly RetryOptions _options;
        private readonly IRetryPolicy _policy;
        private readonly IAuthRefreshHandler? _authRefresh;

        public RetryTransport(
            IStreamingHttpTransport inner,
            RetryOptions? options = null,
            IRetryPolicy? policy = null,
            IAuthRefreshHandler? authRefresh = null)
        {
            _inner = inner ?? throw new ArgumentNullException(nameof(inner));
            _options = options ?? new RetryOptions();
            _policy = policy ?? new ExponentialBackoffRetryPolicy(_options);
            _authRefresh = authRefresh;
        }

        public TransportCapabilities Capabilities => _inner.Capabilities;

        public async Task<ITransportSession> OpenAsync(
            TransportRequest request,
            TransportOptions options,
            CancellationToken cancellationToken)
        {
            int retryCount = 0;
            bool refreshed = false;

            while (true)
            {
                cancellationToken.ThrowIfCancellationRequested();

                try
                {
                    var session = await _inner.OpenAsync(request, options, cancellationToken).ConfigureAwait(false);
                    var failure = ClassifyStatus(session);
                    if (failure == null)
                    {
                        return session;
                    }

                    await session.DisposeAsync().ConfigureAwait(false);
                    throw failure;
                }
                catch (TransportException error)
                {
                    if (error.Kind == TransportErrorKind.Cancelled || error.Kind == TransportErrorKind.Aborted)
                    {
                        throw;
                    }

                    if (error.StatusCode == 401 && _authRefresh != null && !refreshed)
                    {
                        refreshed = true;
                        if (await _authRefresh.RefreshAsync(cancellationToken).ConfigureAwait(false))
                        {
                            continue;
                        }
                        throw;
                    }

                    if (retryCount >= _options.MaxRetries || !_policy.ShouldRetry(error, retryCount))
                    {
                        throw;
                    }

                    var delay = _policy.GetDelay(error, retryCount);
                    if (delay > _options.MaxRetryDelay)
                    {
                        throw;
                    }

                    retryCount++;
                    if (delay > TimeSpan.Zero)
                    {
                        await Task.Delay(delay, cancellationToken).ConfigureAwait(false);
                    }
                }
            }
        }

        private static TransportException? ClassifyStatus(ITransportSession session)
        {
            int status = session.StatusCode;
            if (status >= 200 && status < 300)
            {
                return null;
            }

            bool retryable = status == 408 || status == 409 || status == 429 || status >= 500;
            var headers = session.Headers;
            if (headers.TryGetValue("x-should-retry", out var value) && value != null)
            {
                if (value.Equals("false", StringComparison.OrdinalIgnoreCase))
                {
                    retryable = false;
                }
                else if (value.Equals("true", StringComparison.OrdinalIgnoreCase))
                {
                    retryable = true;
                }
            }

            return new TransportException(
                TransportErrorKind.HttpStatus,
                "HTTP " + status,
                null,
                status,
                headers,
                null,
                retryable);
        }

        private sealed class ExponentialBackoffRetryPolicy : IRetryPolicy
        {
            private readonly RetryOptions _options;
            private readonly Random _random = new Random();

            public ExponentialBackoffRetryPolicy(RetryOptions options)
            {
                _options = options;
            }

            public bool ShouldRetry(TransportException error, int attempt)
            {
                if (error.Headers != null && error.Headers.TryGetValue("x-should-retry", out var value) && value != null)
                {
                    if (value.Equals("false", StringComparison.OrdinalIgnoreCase))
                    {
                        return false;
                    }
                    if (value.Equals("true", StringComparison.OrdinalIgnoreCase))
                    {
                        return true;
                    }
                }
                return error.IsRetryable;
            }

            public TimeSpan GetDelay(TransportException error, int attempt)
            {
                var retryAfter = ParseRetryAfter(error.Headers);
                if (retryAfter.HasValue)
                {
                    return retryAfter.Value;
                }

                var exponential = _options.BaseDelay.TotalMilliseconds * Math.Pow(2, attempt);
                var jitter = _random.NextDouble() * _options.BaseDelay.TotalMilliseconds;
                return TimeSpan.FromMilliseconds(exponential + jitter);
            }

            private static TimeSpan? ParseRetryAfter(TransportHeaders? headers)
            {
                if (headers == null)
                {
                    return null;
                }

                if (headers.TryGetValue("retry-after-ms", out var millisecondsText) &&
                    millisecondsText != null &&
                    int.TryParse(millisecondsText, NumberStyles.Integer, CultureInfo.InvariantCulture, out var milliseconds))
                {
                    return TimeSpan.FromMilliseconds(milliseconds);
                }

                if (headers.TryGetValue("Retry-After", out var secondsText) && secondsText != null)
                {
                    if (int.TryParse(secondsText, NumberStyles.Integer, CultureInfo.InvariantCulture, out var seconds))
                    {
                        return TimeSpan.FromSeconds(seconds);
                    }

                    if (DateTimeOffset.TryParse(
                        secondsText,
                        CultureInfo.InvariantCulture,
                        DateTimeStyles.AssumeUniversal,
                        out var absolute))
                    {
                        var delay = absolute - DateTimeOffset.UtcNow;
                        return delay > TimeSpan.Zero ? delay : TimeSpan.Zero;
                    }
                }

                return null;
            }
        }
    }
}
```

```csharp
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    public interface ITransportRequestFinalizer
    {
        ValueTask<TransportRequest> FinalizeAsync(TransportRequest request, CancellationToken ct);
    }

    /// <summary>管线最内层：紧贴具体 transport，按序执行 finalizer 列表。</summary>
    public sealed class FinalizerTransport : IStreamingHttpTransport
    {
        private readonly IStreamingHttpTransport _inner;
        private readonly IReadOnlyList<ITransportRequestFinalizer> _finalizers;

        public FinalizerTransport(
            IStreamingHttpTransport inner,
            IEnumerable<ITransportRequestFinalizer> finalizers)
        {
            _inner = inner ?? throw new ArgumentNullException(nameof(inner));
            if (finalizers == null)
            {
                throw new ArgumentNullException(nameof(finalizers));
            }
            _finalizers = new List<ITransportRequestFinalizer>(finalizers);
        }

        public TransportCapabilities Capabilities => _inner.Capabilities;

        public async Task<ITransportSession> OpenAsync(
            TransportRequest request,
            TransportOptions options,
            CancellationToken cancellationToken)
        {
            var current = request;
            for (int i = 0; i < _finalizers.Count; i++)
            {
                current = await _finalizers[i].FinalizeAsync(current, cancellationToken).ConfigureAwait(false);
            }
            return await _inner.OpenAsync(current, options, cancellationToken).ConfigureAwait(false);
        }
    }
```

```csharp
using System;
using System.Collections.Generic;

namespace UnityLlmTransport
{
    /// <summary>按 Kind 精确注册；Auto 按 FallbackPolicy 顺序取首个已注册且能力满足的传输。</summary>
    public sealed class DefaultTransportSelector : ITransportSelector
    {
        private readonly IReadOnlyDictionary<TransportKind, IStreamingHttpTransport> _httpTransports;
        private readonly IReadOnlyDictionary<TransportKind, IWebSocketTransport> _webSocketTransports;
        private readonly TransportFallbackPolicy _fallback;

        public DefaultTransportSelector(
            IReadOnlyDictionary<TransportKind, IStreamingHttpTransport> httpTransports,
            IReadOnlyDictionary<TransportKind, IWebSocketTransport>? webSocketTransports = null,
            TransportFallbackPolicy? fallbackPolicy = null)
        {
            _httpTransports = httpTransports ?? throw new ArgumentNullException(nameof(httpTransports));
            _webSocketTransports = webSocketTransports
                ?? new Dictionary<TransportKind, IWebSocketTransport>();
            _fallback = fallbackPolicy
                ?? new TransportFallbackPolicy(new[] { TransportKind.Sse, TransportKind.WebSocket });
        }

        public TransportSelection Select(TransportSelectionContext context)
        {
            if (context == null)
            {
                throw new ArgumentNullException(nameof(context));
            }

            if (context.Kind == TransportKind.WebSocket || context.Kind == TransportKind.WebSocketCached)
            {
                return SelectWebSocket(context.Kind);
            }
            if (context.Kind == TransportKind.Sse)
            {
                return SelectHttp(context.Kind);
            }

            for (int i = 0; i < _fallback.Order.Count; i++)
            {
                var kind = _fallback.Order[i];
                if (kind == TransportKind.WebSocket || kind == TransportKind.WebSocketCached)
                {
                    if (_webSocketTransports.ContainsKey(kind))
                    {
                        return SelectWebSocket(kind);
                    }
                }
                else if (kind == TransportKind.Sse && _httpTransports.ContainsKey(kind))
                {
                    return SelectHttp(kind);
                }
            }

            throw new TransportException(
                TransportErrorKind.Unsupported,
                "No transport registered for auto selection.");
        }

        private TransportSelection SelectHttp(TransportKind kind)
        {
            if (!_httpTransports.TryGetValue(kind, out var transport))
            {
                throw new TransportException(
                    TransportErrorKind.Unsupported,
                    "No HTTP transport registered for " + kind + ".");
            }
            if ((transport.Capabilities & TransportCapabilities.StreamingBody) == 0)
            {
                throw new TransportException(
                    TransportErrorKind.Unsupported,
                    "Registered HTTP transport lacks StreamingBody capability.");
            }
            return new TransportSelection(transport, null);
        }

        private TransportSelection SelectWebSocket(TransportKind kind)
        {
            if (!_webSocketTransports.TryGetValue(kind, out var transport))
            {
                throw new TransportException(
                    TransportErrorKind.Unsupported,
                    "No WebSocket transport registered for " + kind + ".");
            }
            if ((transport.Capabilities & TransportCapabilities.WebSocket) == 0)
            {
                throw new TransportException(
                    TransportErrorKind.Unsupported,
                    "Registered WebSocket transport lacks WebSocket capability.");
            }
            return new TransportSelection(null, transport);
        }
    }
}
```

```csharp
using System;
using System.Collections.Generic;
using System.Net.WebSockets;
using System.Runtime.CompilerServices;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    public sealed class MockWebSocketTransport : IWebSocketTransport
    {
        private readonly Queue<WebSocketMessage> _incoming;
        private readonly List<WebSocketMessage> _sent = new List<WebSocketMessage>();
        private readonly List<WebSocketRequest> _requests = new List<WebSocketRequest>();

        public MockWebSocketTransport(IEnumerable<WebSocketMessage>? incoming = null)
        {
            _incoming = incoming == null
                ? new Queue<WebSocketMessage>()
                : new Queue<WebSocketMessage>(incoming);
        }

        public TransportCapabilities Capabilities => TransportCapabilities.WebSocket;

        public IReadOnlyList<WebSocketMessage> Sent => _sent;
        public IReadOnlyList<WebSocketRequest> Requests => _requests;

        public Task<IWebSocketSession> ConnectAsync(
            WebSocketRequest request,
            WebSocketOptions options,
            CancellationToken cancellationToken)
        {
            _requests.Add(request);
            IWebSocketSession session = new MockWebSocketSession(this, _incoming);
            return Task.FromResult(session);
        }

        private sealed class MockWebSocketSession : IWebSocketSession
        {
            private readonly MockWebSocketTransport _owner;
            private readonly Queue<WebSocketMessage> _incoming;
            private WebSocketState _state = WebSocketState.Open;

            public MockWebSocketSession(MockWebSocketTransport owner, Queue<WebSocketMessage> incoming)
            {
                _owner = owner;
                _incoming = incoming;
            }

            public WebSocketState State => _state;

            public Task SendAsync(WebSocketMessage message, CancellationToken ct)
            {
                _owner._sent.Add(message);
                return Task.CompletedTask;
            }

            public async IAsyncEnumerable<WebSocketMessage> ReceiveAsync(
                [EnumeratorCancellation] CancellationToken ct)
            {
                while (_incoming.Count > 0)
                {
                    ct.ThrowIfCancellationRequested();
                    yield return _incoming.Dequeue();
                    await Task.Yield();
                }
            }

            public Task CloseAsync(WebSocketCloseStatus status, string? description, CancellationToken ct)
            {
                _state = WebSocketState.Closed;
                return Task.CompletedTask;
            }

            public void Abort()
            {
                _state = WebSocketState.Aborted;
            }

            public ValueTask DisposeAsync()
            {
                Abort();
                return default;
            }
        }
    }
}
```
