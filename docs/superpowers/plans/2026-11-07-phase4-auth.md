**Goal:** 在 `UnityLlmTransport.Auth` 中提供传输侧认证/签名公共件（ApiKey/OAuth 注入、SignerFinalizer、能力守卫）与 401-refresh 管线测试；不包含任何消费方集成。

**Tech Stack:** C# 9、netstandard2.1（包）、net10.0 xUnit、Phase 3 `Phase3TestSupport`（仅测试用本地 Kestrel）。

```csharp
namespace UnityLlmTransport
{
    internal static class AuthHeaders
    {
        public static bool ShouldInject(TransportRequest request, string headerName)
        {
            return !request.Headers.ContainsKey(headerName);
        }

        public static TransportRequest WithHeader(TransportRequest request, string headerName, string value)
        {
            var headers = new TransportHeaders();
            foreach (var pair in request.Headers)
            {
                headers[pair.Key] = pair.Value;
            }
            headers[headerName] = value;
            return request with { Headers = headers };
        }
    }
}
```

`src/UnityLlmTransport.Auth/ApiKeyMiddleware.cs`：

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    public sealed record ApiKeyMiddlewareOptions
    {
        public string HeaderName { get; init; } = "Authorization";
        public string? Scheme { get; init; } = "Bearer";
    }

    public sealed class ApiKeyMiddleware : IRequestMiddleware
    {
        private readonly string _apiKey;
        private readonly ApiKeyMiddlewareOptions _options;

        public ApiKeyMiddleware(string apiKey, ApiKeyMiddlewareOptions? options = null)
        {
            if (string.IsNullOrEmpty(apiKey))
            {
                throw new ArgumentException("API key must not be null or empty.", nameof(apiKey));
            }
            _apiKey = apiKey;
            _options = options ?? new ApiKeyMiddlewareOptions();
            if (string.IsNullOrEmpty(_options.HeaderName))
            {
                throw new ArgumentException("HeaderName must not be null or empty.", nameof(options));
            }
        }

        public ValueTask<TransportRequest> InvokeAsync(TransportRequest request, CancellationToken ct)
        {
            if (request == null)
            {
                throw new ArgumentNullException(nameof(request));
            }
            if (!AuthHeaders.ShouldInject(request, _options.HeaderName))
            {
                return new ValueTask<TransportRequest>(request);
            }

            var value = string.IsNullOrEmpty(_options.Scheme)
                ? _apiKey
                : _options.Scheme + " " + _apiKey;
            return new ValueTask<TransportRequest>(
                AuthHeaders.WithHeader(request, _options.HeaderName, value));
        }
    }
}
```

```csharp
using System;
using System.Threading;
using System.Threading.Tasks;

namespace UnityLlmTransport
{
    public sealed record OAuthTokenMiddlewareOptions
    {
        public string HeaderName { get; init; } = "Authorization";
        public string Scheme { get; init; } = "Bearer";
    }

    public sealed class OAuthTokenMiddleware : IRequestMiddleware
    {
        private readonly Func<CancellationToken, ValueTask<string>> _tokenProvider;
        private readonly OAuthTokenMiddlewareOptions _options;

        public OAuthTokenMiddleware(
            Func<CancellationToken, ValueTask<string>> tokenProvider,
            OAuthTokenMiddlewareOptions? options = null)
        {
            _tokenProvider = tokenProvider ?? throw new ArgumentNullException(nameof(tokenProvider));
            _options = options ?? new OAuthTokenMiddlewareOptions();
            if (string.IsNullOrEmpty(_options.HeaderName))
            {
                throw new ArgumentException("HeaderName must not be null or empty.", nameof(options));
            }
        }

        public async ValueTask<TransportRequest> InvokeAsync(TransportRequest request, CancellationToken ct)
        {
            if (request == null)
            {
                throw new ArgumentNullException(nameof(request));
            }
            if (!AuthHeaders.ShouldInject(request, _options.HeaderName))
            {
                return request;
            }

            string token = await _tokenProvider(ct).ConfigureAwait(false);
            if (string.IsNullOrEmpty(token))
            {
                throw new TransportException(
                    TransportErrorKind.Protocol,
                    "OAuth token provider returned an empty token.");
            }

            var value = string.IsNullOrEmpty(_options.Scheme)
                ? token
                : _options.Scheme + " " + token;
            return AuthHeaders.WithHeader(request, _options.HeaderName, value);
        }
    }
}
```

```csharp
namespace UnityLlmTransport
```

```csharp
        private static async Task<string> ReadAllTextAsync(Stream stream, CancellationToken ct)
        {
            var buffer = new byte[256];
            var builder = new StringBuilder();
            int read;
            while ((read = await stream.ReadAsync(buffer.AsMemory(), ct)) > 0)
            {
                builder.Append(Encoding.UTF8.GetString(buffer, 0, read));
            }
            return builder.ToString();
        }

        private sealed class DelegateRefreshHandler : IAuthRefreshHandler
        {
            private readonly Action _onRefresh;

            public DelegateRefreshHandler(Action onRefresh)
            {
                _onRefresh = onRefresh;
            }

            public Task<bool> RefreshAsync(CancellationToken ct)
            {
                _onRefresh();
                return Task.FromResult(true);
            }
        }

        private sealed class ZeroDelayPolicy : IRetryPolicy
        {
            public bool ShouldRetry(TransportException error, int attempt)
            {
                return error.IsRetryable;
            }

            public TimeSpan GetDelay(TransportException error, int attempt)
            {
                return TimeSpan.Zero;
            }
        }

        private sealed class HeaderSigner : IRequestSigner
        {
            private readonly string _name;
            private readonly string _value;

            public HeaderSigner(string name, string value)
            {
                _name = name;
                _value = value;
            }

            public ValueTask<TransportRequest> SignAsync(TransportRequest request, CancellationToken ct)
            {
                var headers = new TransportHeaders();
                foreach (var pair in request.Headers)
                {
                    headers[pair.Key] = pair.Value;
                }
                headers[_name] = _value;
                return new ValueTask<TransportRequest>(request with { Headers = headers });
            }
        }
```
